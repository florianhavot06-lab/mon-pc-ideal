function openAffiliateModal() {
  const modal = document.getElementById('affiliate-modal');
  const inputAmazon = document.getElementById('input-amazon-tag');
  const inputSub = document.getElementById('input-subid-tag');
  if (inputAmazon) inputAmazon.value = affiliateConfig.amazonTag || '';
  if (inputSub) inputSub.value = affiliateConfig.subId || '';
  if (modal) modal.classList.add('open');
}

function closeAffiliateModal() {
  const modal = document.getElementById('affiliate-modal');
  if (modal) modal.classList.remove('open');
}

function saveAffiliateSettings() {
  const inputAmazon = document.getElementById('input-amazon-tag');
  const inputSub = document.getElementById('input-subid-tag');

  affiliateConfig.amazonTag = inputAmazon ? inputAmazon.value.trim() : '';
  affiliateConfig.subId = inputSub ? inputSub.value.trim() : '';

  try {
    localStorage.setItem('dazl-affiliate-config', JSON.stringify(affiliateConfig));
  } catch (e) {}

  if (window.smSetUrlParam) {
    if (affiliateConfig.amazonTag) window.smSetUrlParam('amz_tag', affiliateConfig.amazonTag);
    if (affiliateConfig.subId) window.smSetUrlParam('sub_id', affiliateConfig.subId);
  }

  updateAffiliateUIState();
  closeAffiliateModal();
  recalculateAll();

  const toast = document.getElementById('toast-notify');
  if (toast) {
    toast.textContent = "Tags d'affiliation enregistrés et activés !";
    toast.style.display = 'block';
    setTimeout(() => { toast.style.display = 'none'; }, 2600);
  }
}

  const inputAmazon = document.getElementById('input-amazon-tag');
  const inputSub = document.getElementById('input-subid-tag');

function openComponentPicker(cat) {
  if (!HARDWARE_DB[cat]) return;
  activePickerCat = cat;
  activePickerFilter = "Tous";
  activePickerSearch = "";

  const modal = document.getElementById('comp-picker-modal');
  const titleText = document.getElementById('picker-title-text');
  const subText = document.getElementById('picker-modal-sub');
  const searchInput = document.getElementById('picker-search-input');
  const clearBtn = document.getElementById('picker-search-clear');

  const catLabel = CATEGORY_NAMES[cat] || cat;
  const count = HARDWARE_DB[cat].length;
  const vConfig = VENDORS[preferredMerchant] || VENDORS.amazon;

  renderPickerItems();

function closeComponentPicker() {
  const modal = document.getElementById('comp-picker-modal');

function renderPickerFilterPills() {

function matchItemWithFilter(item, filter, cat) {
  if (filter === "Tous") return true;
  const str = `${item.name} ${item.spec || ''} ${item.socket || ''} ${item.ramType || ''}`.toLowerCase();

  if (cat === 'cpu') {
    if (filter === "AMD AM5") return item.socket === 'AM5';
    if (filter === "AMD AM4") return (item.socket === 'AM4' || item.socketExact === 'AM4');
    if (filter === "Intel LGA1700") return item.socket === 'LGA1700';
    if (filter === "Intel LGA1851") return item.socket === 'LGA1851';
    if (filter === "3D V-Cache") return str.includes('x3d') || str.includes('v-cache');
  } else if (cat === 'gpu') {
    if (filter === "NVIDIA RTX") return str.includes('rtx');
    if (filter === "AMD Radeon") return str.includes('rx') || str.includes('radeon');
    if (filter === "Intel Arc") return str.includes('arc') || str.includes('b580');
    if (filter === "4K Ready") return item.tier >= 4;
  } else if (cat === 'mobo') {
    if (filter === "AMD AM5") return item.socket === 'AM5';
    if (filter === "AMD AM4") return item.socket === 'AM4';
    if (filter === "Intel LGA1700") return item.socket === 'LGA1700';
    if (filter === "Intel LGA1851") return item.socket === 'LGA1851';
    if (filter === "DDR5") return item.ramType === 'DDR5';
    if (filter === "DDR4") return item.ramType === 'DDR4';
  } else if (cat === 'ram') {
    if (filter === "DDR5") return item.ramType === 'DDR5';
    if (filter === "DDR4") return item.ramType === 'DDR4';
    if (filter === "32 Go") return str.includes('32 go') || str.includes('32gb');
    if (filter === "64 Go+") return str.includes('64 go') || str.includes('96 go') || str.includes('48 go');
    if (filter === "RGB") return str.includes('rgb');
  } else if (cat === 'ssd') {
    if (filter === "1 To") return str.includes('1 to');
    if (filter === "2 To") return str.includes('2 to');
    if (filter === "4 To") return str.includes('4 to');
    if (filter === "Gen4") return str.includes('gen4');
    if (filter === "Gen5") return str.includes('gen5');
  } else if (cat === 'cooler') {
    if (filter === "Ventirad Air") return !str.includes('aio') && !str.includes('watercooling');
    if (filter === "AIO 240mm") return str.includes('240');
    if (filter === "AIO 360mm") return str.includes('360');
    if (filter === "AIO 420mm") return str.includes('420');
    if (filter === "Écran LCD") return str.includes('lcd') || str.includes('oled');
  } else if (cat === 'psu') {
    if (filter === "550-650W") return item.watts <= 650;
    if (filter === "750-850W") return item.watts >= 750 && item.watts <= 850;
    if (filter === "1000W+") return item.watts >= 1000;
    if (filter === "ATX 3.0") return str.includes('atx 3.0') || str.includes('pcie5');
    if (filter === "Titanium/Platinum") return str.includes('platinum') || str.includes('titanium');
  } else if (cat === 'case') {
    if (filter === "Airflow Mesh") return str.includes('airflow') || str.includes('mesh') || str.includes('air');
    if (filter === "Aquarium Panoramique") return str.includes('panoramique') || str.includes('aquarium') || str.includes('fishbowl') || str.includes('o11');
    if (filter === "Compact SFF") return str.includes('compact') || str.includes('sff') || str.includes('matx');
    if (filter === "Façade Bois") return str.includes('bois') || str.includes('north');
  } else if (cat === 'monitor') {
    if (filter === "1080p Esport") return item.res === '1080p';
    if (filter === "1440p QHD") return item.res === '1440p';
    if (filter === "4K UHD") return item.res === '4k';
    if (filter === "OLED") return str.includes('oled');
    if (filter === "Ultrawide") return str.includes('ultrawide') || str.includes('21:9') || str.includes('32:9');
  } else if (cat === 'keyboard') {
    if (filter === "Mécanique") return str.includes('mécanique');
    if (filter === "Esport Pro") return str.includes('rapid trigger') || str.includes('omnipoint') || str.includes('optique');
    if (filter === "Membrane") return str.includes('membrane') || str.includes('mech-dome');
  } else if (cat === 'mouse') {
    if (filter === "Sans Fil") return str.includes('sans fil') || str.includes('lightspeed');
    if (filter === "Ultra-légère (<60g)") return str.includes('54g') || str.includes('59g') || str.includes('60g');
    if (filter === "Filaire") return str.includes('filaire') || str.includes('câblée');
  } else if (cat === 'headset') {
    if (filter === "Sans Fil") return str.includes('sans fil') || str.includes('wireless') || str.includes('lightspeed');
    if (filter === "Confort Gaming") return str.includes('cloud') || str.includes('arctis');
    if (filter === "Studio Audiophile") return str.includes('audiophile') || str.includes('studio') || str.includes('mmx');
  } else if (cat === 'os') {
    if (filter === "Licences Windows") return str.includes('windows');
    if (filter === "Montage Atelier") return str.includes('montage') || str.includes('prestation');
  }

  return str.includes(filter.toLowerCase());
}

function renderPickerItems() {
  const container = document.getElementById('picker-items-list');
  if (!container || !activePickerCat) return;

  const items = HARDWARE_DB[activePickerCat] || [];
  const vConfig = VENDORS[preferredMerchant] || VENDORS.amazon;
  const currentId = currentSelection[activePickerCat];

  const filtered = items.filter(item => {
    const matchesFilter = matchItemWithFilter(item, activePickerFilter, activePickerCat);
    if (!matchesFilter) return false;
    if (!activePickerSearch) return true;
    const searchTarget = `${item.name} ${item.spec || ''} ${item.socket || ''} ${item.ramType || ''} ${item.vendor || ''}`.toLowerCase();
    return searchTarget.includes(activePickerSearch);
  });

  if (filtered.length === 0) {
    container.innerHTML = `
      <div class="picker-empty-state">
        <p>Aucun composant ne correspond à vos critères de recherche.</p>
        <button type="button" class="btn" style="margin-top:10px;" onclick="clearPickerSearch(); setPickerFilter('Tous');">Réinitialiser les filtres</button>
      </div>
    `;
    return;
  }

  container.innerHTML = filtered.map(item => {
    const isSelected = item.id === currentId;
    const price = getMerchantItemPrice(item.price, preferredMerchant, activePickerCat);
    const priceText = item.price === 0 ? "Gratuit / Inclus" : `${price} €`;

    const rev = getComponentReview(activePickerCat, item.id);
    let tagsHtml = '';
    tagsHtml += `<span class="picker-tag" style="color:#fbbf24; cursor:pointer; border-color:rgba(245, 158, 11, 0.35); background:rgba(245, 158, 11, 0.08);" onclick="event.stopPropagation(); openReviewModal('${activePickerCat}', '${item.id}')" title="Lire l'avis complet">★ ${rev.rating} Avis</span>`;
    if (item.spec) tagsHtml += `<span class="picker-tag">${item.spec}</span>`;
    if (item.tdp) tagsHtml += `<span class="picker-tag tdp">${item.tdp}W TDP</span>`;
    if (item.watts) tagsHtml += `<span class="picker-tag tdp">${item.watts}W Capacité</span>`;

    return `
      <div class="picker-item-row ${isSelected ? 'selected' : ''}" onclick="selectItemFromPicker('${item.id}')">
        <div class="picker-item-main">
          <div class="picker-item-title-row">
            <span class="picker-item-name">${item.name}</span>
            ${isSelected ? '<span class="picker-badge-selected">Installé ✓</span>' : ''}
          </div>
          <div class="picker-item-tags">
            ${tagsHtml}
          </div>
        </div>
        <div class="picker-item-price-wrap">
          <span class="picker-item-price">${priceText}</span>
          <span class="picker-item-vendor">chez ${vConfig.name}</span>
        </div>
      </div>
    `;
  }).join('');
}

function selectItemFromPicker(itemId) {
  if (!activePickerCat) return;
  onComponentChange(activePickerCat, itemId);
  updateSelectElements();
  closeComponentPicker();
}

function initSelectOptions() {
  const categories = Object.keys(HARDWARE_DB);
  categories.forEach(cat => {
    const sel = document.getElementById(`select-${cat}`);
    if (!sel) return;
    const currentVal = currentSelection[cat] || sel.value;
    sel.innerHTML = HARDWARE_DB[cat].map(item => {
      const extra = item.watts ? ` (${item.watts}W)` : (item.spec ? ` (${item.spec.split('•')[0].trim()})` : '');
      return `<option value="${item.id}">${item.name}${extra}</option>`;
    }).join('');
    if (currentVal) sel.value = currentVal;

    // Attach custom trigger button to wrap
    const wrap = sel.parentElement;
    if (wrap) {
      wrap.onclick = (e) => {
        if (e.target.tagName !== 'SELECT') openComponentPicker(cat);
      };
      wrap.title = "Cliquer pour parcourir le catalogue et rechercher";
      wrap.style.cursor = "pointer";
      sel.style.display = "none";

      let trigger = wrap.querySelector('.custom-select-trigger');
      if (!trigger) {
        trigger = document.createElement('div');
        trigger.className = 'custom-select-trigger';
        trigger.id = `trigger-${cat}`;
        trigger.innerHTML = `
          <div class="trigger-text-wrap">
            <span class="trigger-name" id="trigger-name-${cat}"></span>
            <span class="trigger-spec" id="trigger-spec-${cat}"></span>
          </div>
          <div class="trigger-action">
            <span class="trigger-browse-btn">Changer</span>
            <svg class="trigger-chevron" width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><polyline points="6 9 12 15 18 9"></polyline></svg>
          </div>
        `;
        wrap.insertBefore(trigger, sel);
      }
    }
  });
}

function updateSelectElements() {
  Object.keys(currentSelection).forEach(cat => {
    const sel = document.getElementById(`select-${cat}`);
    if (sel) sel.value = currentSelection[cat];
    const item = HARDWARE_DB[cat]?.find(i => i.id === currentSelection[cat]);
    const trgName = document.getElementById(`trigger-name-${cat}`);
    const trgSpec = document.getElementById(`trigger-spec-${cat}`);
    if (trgName && item) trgName.textContent = item.name;
    if (trgSpec && item) trgSpec.textContent = item.spec || '';
  });
}

function switchView(view) {
  const btnBuilder = document.getElementById('tab-btn-builder');
  const btnReviews = document.getElementById('tab-btn-reviews');
  const btnComparator = document.getElementById('tab-btn-comparator');
  const btnGuide = document.getElementById('tab-btn-guide');
  const viewBuilder = document.getElementById('view-builder');
  const viewReviews = document.getElementById('view-reviews');
  const viewComparator = document.getElementById('view-comparator');
  const viewGuide = document.getElementById('view-guide');

  [btnBuilder, btnReviews, btnComparator, btnGuide].forEach(b => {
    if (b) {
      b.classList.remove('active');
      b.setAttribute('aria-selected', 'false');
    }
  });
  if (viewBuilder) viewBuilder.style.display = 'none';
  if (viewReviews) viewReviews.classList.remove('active');
  if (viewComparator) viewComparator.classList.remove('active');
  if (viewGuide) viewGuide.classList.remove('active');

  if (view === 'builder') {
    if (btnBuilder) {
      btnBuilder.classList.add('active');
      btnBuilder.setAttribute('aria-selected', 'true');
    }
    if (viewBuilder) viewBuilder.style.display = 'grid';
    if (window.smSetUrlParam) window.smSetUrlParam('tab', 'builder');
    setTimeout(() => {
      renderFpsChart();
      renderBudgetChart();
    }, 50);
  } else if (view === 'reviews') {
    if (btnReviews) {
      btnReviews.classList.add('active');
      btnReviews.setAttribute('aria-selected', 'true');
    }
    if (viewReviews) viewReviews.classList.add('active');
    if (window.smSetUrlParam) window.smSetUrlParam('tab', 'reviews');
    renderReviewsView();
  } else if (view === 'comparator') {
    if (btnComparator) {
      btnComparator.classList.add('active');
      btnComparator.setAttribute('aria-selected', 'true');
    }
    if (viewComparator) viewComparator.classList.add('active');
    if (window.smSetUrlParam) window.smSetUrlParam('tab', 'comparator');
    renderPriceComparisonTable();
  } else {
    if (btnGuide) {
      btnGuide.classList.add('active');
      btnGuide.setAttribute('aria-selected', 'true');
    }
    if (viewGuide) viewGuide.classList.add('active');
    if (window.smSetUrlParam) window.smSetUrlParam('tab', 'guide');
  }
}

function getMerchantItemPrice(basePrice, vendorKey, cat) {
  if (basePrice === 0) return 0;
  // Specific category advantages per store
  let factor = 1.0;
  if (vendorKey === 'amazon') {
    factor = (cat === 'ssd' || cat === 'ram') ? 0.96 : (cat === 'gpu' ? 0.99 : 1.0);
  } else if (vendorKey === 'ldlc') {
    factor = (cat === 'mobo' || cat === 'psu') ? 1.02 : 1.03;
  } else if (vendorKey === 'topachat') {
    factor = (cat === 'gpu' || cat === 'case') ? 0.98 : 1.01;
  } else if (vendorKey === 'alternate') {
    factor = (cat === 'cpu' || cat === 'cooler') ? 0.95 : 0.99;
  }
  return Math.round(basePrice * factor);
}

function getCoolerThermal(cooler) {
  if (!cooler) return { capacity: 220, noiseLoad: 33, noiseIdle: 20 };
  const id = cooler.id || '';
  if (id.includes('360') || id.includes('kraken') || id.includes('ryujin') || id.includes('icue')) {
    return { capacity: 350, noiseLoad: 31, noiseIdle: 19 };
  }
  if (id.includes('240') || id.includes('280') || id.includes('aqua')) {
    return { capacity: 260, noiseLoad: 34, noiseIdle: 22 };
  }
  if (id.includes('pa120') || id.includes('ps120') || id.includes('d15')) {
    return { capacity: 250, noiseLoad: 32, noiseIdle: 20 };
  }
  return { capacity: 180, noiseLoad: 35, noiseIdle: 22 };
}

function getCaseThermal(pcCase) {
  if (!pcCase) return { airflowDelta: -2, noiseAdd: 0 };
  const id = pcCase.id || '';
  if (id.includes('airflow') || id.includes('pop') || id.includes('216') || id.includes('903') || id.includes('cc560') || id.includes('north')) {
    return { airflowDelta: -3, noiseAdd: 1 };
  }
  if (id.includes('fishbowl') || id.includes('lianli') || id.includes('c8') || id.includes('y70')) {
    return { airflowDelta: -1, noiseAdd: 1 };
  }
  return { airflowDelta: -1, noiseAdd: 0 };
}

function updateThermalSimulation(selectedItems) {
  const cpu = selectedItems.cpu || HARDWARE_DB.cpu[0];
  const gpu = selectedItems.gpu || HARDWARE_DB.gpu[0];
  const cooler = selectedItems.cooler || HARDWARE_DB.cooler[0];
  const pcCase = selectedItems.case || HARDWARE_DB.case[0];

  const coolerSpec = getCoolerThermal(cooler);
  const caseSpec = getCaseThermal(pcCase);

  let cpuTemp, gpuTemp, totalDb;
  let cpuNote = "";
  let gpuNote = "";
  let noiseNote = "";

  if (thermalMode === 'idle') {
    cpuTemp = Math.round(33 + (cpu.tdp / 50) + (caseSpec.airflowDelta * 0.5));
    gpuTemp = Math.round(34 + (caseSpec.airflowDelta * 0.4));
    totalDb = Math.round(coolerSpec.noiseIdle + caseSpec.noiseAdd);

    cpuNote = "Mode Repos : fréquences basses, ventilateurs discrets";
    gpuNote = "Mode 0 dB Fan-Stop actif (ventilateurs coupés)";
    noiseNote = "Quasi inaudible dans un environnement calme";
  } else {
    // Gaming Load simulation
    const cpuHeatRatio = cpu.tdp / coolerSpec.capacity;
    cpuTemp = Math.round(48 + (cpuHeatRatio * 44) + caseSpec.airflowDelta);
    if (cpuTemp > 95) cpuTemp = 95;

    const gpuBase = 58 + Math.round((gpu.tdp / 550) * 16);
    gpuTemp = Math.round(gpuBase + caseSpec.airflowDelta);

    // Acoustique combinée (dB)
    const gpuDb = 28 + Math.round((gpu.tdp / 550) * 10);
    const coolerDb = coolerSpec.noiseLoad;
    const maxPart = Math.max(gpuDb, coolerDb);
    const minPart = Math.min(gpuDb, coolerDb);
    totalDb = Math.round(maxPart + (10 * Math.log10(1 + Math.pow(10, (minPart - maxPart) / 10))) + caseSpec.noiseAdd);

    if (cpuTemp >= 88) {
      cpuNote = "Risque de bridage thermique (Thermal Throttling) !";
    } else if (cpuTemp >= 78) {
      cpuNote = "Chauffe soutenue, ventilation active";
    } else {
      cpuNote = "Températures très confortables, forte marge";
    }

    if (gpuTemp >= 80) {
      gpuNote = "Chauffe modérée, boîtier bien sollicité";
    } else {
      gpuNote = "Refroidissement fluide & stable";
    }

    if (totalDb <= 29) {
      noiseNote = "Configuration ultra-silencieuse même en jeu";
    } else if (totalDb <= 36) {
      noiseNote = "Bruit feutré, parfaitement tolérable sous le bureau";
    } else {
      noiseNote = "Ventilation soutenue audible avec un casque ouvert";
    }
  }

  // Update DOM metrics
  const valCpu = document.getElementById('val-cpu-temp');
  const barCpu = document.getElementById('bar-cpu-temp');
  const noteCpu = document.getElementById('note-cpu-thermal');
  if (valCpu) {
    valCpu.textContent = `${cpuTemp} °C`;
    valCpu.className = `thermal-metric-value ${cpuTemp >= 85 ? 'danger' : (cpuTemp >= 76 ? 'warning' : 'ok')}`;
  }
  if (barCpu) {
    barCpu.style.width = `${Math.min(100, Math.round((cpuTemp / 100) * 100))}%`;
    barCpu.style.backgroundColor = cpuTemp >= 85 ? '#ef4444' : (cpuTemp >= 76 ? '#f59e0b' : '#10b981');
  }
  if (noteCpu) noteCpu.textContent = cpuNote;

  const valGpu = document.getElementById('val-gpu-temp');
  const barGpu = document.getElementById('bar-gpu-temp');
  const noteGpu = document.getElementById('note-gpu-thermal');
  if (valGpu) {
    valGpu.textContent = `${gpuTemp} °C`;
    valGpu.className = `thermal-metric-value ${gpuTemp >= 80 ? 'danger' : (gpuTemp >= 72 ? 'warning' : 'ok')}`;
  }
  if (barGpu) {
    barGpu.style.width = `${Math.min(100, Math.round((gpuTemp / 100) * 100))}%`;
    barGpu.style.backgroundColor = gpuTemp >= 80 ? '#ef4444' : (gpuTemp >= 72 ? '#f59e0b' : '#10b981');
  }
  if (noteGpu) noteGpu.textContent = gpuNote;

  const valDb = document.getElementById('val-noise-db');
  const barDb = document.getElementById('bar-noise-db');
  const noteNoise = document.getElementById('note-noise-acoustic');
  if (valDb) valDb.textContent = `${totalDb} dB(A)`;
  if (barDb) {
    const dbPercent = Math.max(5, Math.min(95, Math.round(((totalDb - 20) / 30) * 100)));
    barDb.style.left = `${dbPercent}%`;
  }
  if (noteNoise) noteNoise.textContent = noiseNote;

  return { cpuTemp, gpuTemp, totalDb };
}

function recalculateAll() {
  let totalPrice = 0;
  let totalWatts = 50;

  const vConfig = VENDORS[preferredMerchant] || VENDORS.amazon;
  const selectedItems = {};
  Object.keys(currentSelection).forEach(cat => {
    const item = HARDWARE_DB[cat].find(i => i.id === currentSelection[cat]);
    selectedItems[cat] = item;
    if (item) {
      const merchantPrice = getMerchantItemPrice(item.price, preferredMerchant, cat);
      totalPrice += merchantPrice;
      if (item.tdp) totalWatts += item.tdp;

      const priceEl = document.getElementById(`price-${cat}`);
      const specEl = document.getElementById(`spec-${cat}`);
      if (priceEl) priceEl.textContent = `${merchantPrice} €`;
      if (specEl && item.spec) specEl.textContent = item.spec;

      const rev = getComponentReview(cat, item.id);
      const revScoreEl = document.getElementById(`lbl-review-${cat}`);
      if (revScoreEl) revScoreEl.textContent = rev.rating;

      const buyBtn = document.getElementById(`btn-buy-${cat}`);
      const vendorBadge = document.getElementById(`vendor-${cat}`);
      if (buyBtn) {
        if (item.price === 0) {
          buyBtn.style.display = 'none';
        } else {
          buyBtn.style.display = 'inline-flex';
          const targetUrl = vConfig.getUrl(item.name);
          buyBtn.href = targetUrl;
          buyBtn.setAttribute('title', `Rechercher ${item.name} chez ${vConfig.name}`);
          if (vendorBadge) vendorBadge.textContent = vConfig.name;
        }
      }
    }
  });

  // KPI Total Price
  const kpiPrice = document.getElementById('kpi-total-price');
  if (kpiPrice) kpiPrice.textContent = `${totalPrice.toLocaleString('fr-FR')} €`;
  const kpiSub = document.getElementById('kpi-price-sub');
  if (kpiSub) kpiSub.textContent = `Tarifs chez ${vConfig.name} TTC`;
  // Target Budget Calculation & Gauge
  const budgetInput = document.getElementById('input-target-budget');
  if (budgetInput) {
    const currentBudVal = parseInt(budgetInput.value, 10);
    if (!isNaN(currentBudVal) && currentBudVal > 0) targetBudget = currentBudVal;
  }

  const budgetDiff = targetBudget - totalPrice;
  const budgetPercent = Math.round((totalPrice / targetBudget) * 100);
  const budgetFill = document.getElementById('budget-progress-fill');
  const budgetBadge = document.getElementById('budget-status-badge');
  const budgetRemText = document.getElementById('budget-remaining-text');

  if (budgetFill) {
    budgetFill.style.width = `${Math.min(100, budgetPercent)}%`;
    if (budgetPercent > 100) {
      budgetFill.style.backgroundColor = '#ef4444';
    } else if (budgetPercent >= 90) {
      budgetFill.style.backgroundColor = '#f59e0b';
    } else {
      budgetFill.style.backgroundColor = '#10b981';
    }
  }

  if (budgetBadge && budgetRemText) {
    if (budgetDiff < 0) {
      budgetBadge.className = 'budget-status-pill over-budget';
      budgetBadge.textContent = `⚠ Dépassement (+${Math.abs(budgetDiff).toLocaleString('fr-FR')} €)`;
      budgetRemText.textContent = `Excédent : +${Math.abs(budgetDiff).toLocaleString('fr-FR')} € (${budgetPercent}%)`;
    } else if (budgetPercent >= 90) {
      budgetBadge.className = 'budget-status-pill near-budget';
      budgetBadge.textContent = `⚡ Seuil atteint (${budgetPercent}%)`;
      budgetRemText.textContent = `Reste : ${budgetDiff.toLocaleString('fr-FR')} €`;
    } else {
      budgetBadge.className = 'budget-status-pill in-budget';
      budgetBadge.textContent = `✓ Dans le budget (${budgetPercent}%)`;
      budgetRemText.textContent = `Reste : ${budgetDiff.toLocaleString('fr-FR')} €`;
    }
  }

  // Power Consumption & PSU
  const psuCapacity = selectedItems.psu ? selectedItems.psu.watts : 750;
  const kpiPower = document.getElementById('kpi-total-power');
  if (kpiPower) kpiPower.textContent = `${totalWatts} W`;

  const psuLoad = Math.round((totalWatts / psuCapacity) * 100);
  const psuBar = document.getElementById('psu-bar-fill');
  const psuLoadLabel = document.getElementById('psu-load-percent');
  const lblConso = document.getElementById('lbl-conso-draw');
  const lblCap = document.getElementById('lbl-psu-capacity');

  if (lblConso) lblConso.textContent = `Conso : ${totalWatts} W`;
  if (lblCap) lblCap.textContent = `Capacité : ${psuCapacity} W (Marge : ${psuCapacity - totalWatts} W)`;
  if (psuLoadLabel) psuLoadLabel.textContent = `${psuLoad}%`;

  if (psuBar) {
    psuBar.style.width = `${Math.min(psuLoad, 100)}%`;
    if (psuLoad > 90) {
      psuBar.className = 'power-bar-fill overflow';
    } else {
      psuBar.className = 'power-bar-fill';
    }
  }

  // Update Thermal & Acoustics
  const thermalResults = updateThermalSimulation(selectedItems);

  // Compatibility Checks
  const alerts = [];
  let isSocketOk = true;
  if (selectedItems.cpu && selectedItems.mobo) {
    if (selectedItems.cpu.socket !== selectedItems.mobo.socket) {
      alerts.push(`Incompatibilité socket : Processeur ${selectedItems.cpu.socket} non supporté par la carte mère (${selectedItems.mobo.socket}) !`);
      isSocketOk = false;
    }
  }

  let isRamOk = true;
  if (selectedItems.mobo && selectedItems.ram) {
    if (selectedItems.mobo.ramType && selectedItems.ram.ramType && selectedItems.mobo.ramType !== selectedItems.ram.ramType) {
      alerts.push(`Incompatibilité mémoire : Carte mère (${selectedItems.mobo.ramType}) incompatible avec la RAM (${selectedItems.ram.ramType}) !`);
      isRamOk = false;
    }
  }

  let isPsuOk = true;
  if (totalWatts > psuCapacity) {
    alerts.push(`Alimentation sous-dimensionnée : Votre système requiert ~${totalWatts}W mais l'alimentation délivre ${psuCapacity}W !`);
    isPsuOk = false;
  } else if (psuLoad > 85) {
    alerts.push(`Attention : L'alimentation fonctionne à ${psuLoad}% de sa capacité. Une marge supérieure est recommandée.`);
  }

  if (thermalResults && thermalResults.cpuTemp >= 88) {
    alerts.push(`Dissipation thermique critique : Le processeur risque de saturer le refroidisseur en forte charge.`);
  }
  if (budgetDiff < 0) {
    alerts.push(`Plafond de budget dépassé : Votre sélection actuelle dépasse l'objectif fixé de ${Math.abs(budgetDiff).toLocaleString('fr-FR')} €.`);
  }

  const alertBox = document.getElementById('compat-alert-box');
  const alertText = document.getElementById('compat-alert-text');
  const compatStatus = document.getElementById('kpi-compat-status');
  const compatNote = document.getElementById('kpi-compat-note');

  if (!isSocketOk || !isPsuOk || !isRamOk) {
    alertBox.className = 'status-alert error';
    alertText.textContent = alerts.join(' | ');
    compatStatus.textContent = 'Erreur';
    compatStatus.style.color = '#ef4444';
    compatNote.textContent = 'Conflit matériel détecté';
  } else if (alerts.length > 0) {
    alertBox.className = 'status-alert warning';
    alertText.textContent = alerts.join(' | ');
    compatStatus.textContent = 'Attention';
    compatStatus.style.color = '#f59e0b';
    compatNote.textContent = 'Marge faible mais fonctionnel';
  } else {
    alertBox.className = 'status-alert ok';
    alertText.textContent = `Compatibilité parfaite : Socket ${selectedItems.cpu.socket} et mémoire ${selectedItems.ram.ramType} alignés, ${psuCapacity - totalWatts}W de marge.`;
    compatStatus.textContent = '100% OK';
    compatStatus.style.color = '#10b981';
    compatNote.textContent = 'Socket, RAM & Wattage validés';
  }

  // Bottleneck estimation
  const cpuTier = selectedItems.cpu ? selectedItems.cpu.tier : 3;
  const gpuTier = selectedItems.gpu ? selectedItems.gpu.tier : 3;
  const kpiBottleneck = document.getElementById('kpi-bottleneck');

  const diff = gpuTier - cpuTier;
  if (diff >= 3) {
    kpiBottleneck.textContent = 'GPU Bridé';
    kpiBottleneck.style.color = '#f59e0b';
  } else if (diff <= -2) {
    kpiBottleneck.textContent = 'CPU Dominant';
    kpiBottleneck.style.color = '#c89650';
  } else {
    kpiBottleneck.textContent = 'Parfaitement Équilibré';
    kpiBottleneck.style.color = '#10b981';
  }

  renderFpsChart();
  renderBudgetChart();
  renderPriceComparisonTable();
  const viewReviews = document.getElementById('view-reviews');
  if (viewReviews && viewReviews.classList.contains('active')) {
    renderReviewsView();
  }
}

function renderFpsChart() {
  const container = document.getElementById('fps-chart-container');
  if (!container || container.clientWidth === 0) return;

  const resolutionSelect = document.getElementById('select-resolution');
  const resolution = resolutionSelect ? resolutionSelect.value : '1440p';

  const gpuItem = HARDWARE_DB.gpu.find(g => g.id === currentSelection.gpu) || HARDWARE_DB.gpu[2];
  const cpuItem = HARDWARE_DB.cpu.find(c => c.id === currentSelection.cpu) || HARDWARE_DB.cpu[1];

  let baseFps = 100;
  if (resolution === '1080p') baseFps = gpuItem.perf1080;
  else if (resolution === '1440p') baseFps = gpuItem.perf1440;
  else if (resolution === '4k') baseFps = gpuItem.perf4k;

  const cpuMultiplier = 0.8 + (cpuItem.tier * 0.05);

  const data = BENCHMARK_GAMES.map(g => {
    const computed = Math.round(baseFps * g.factor * (resolution === '1080p' ? cpuMultiplier : 1.0));
    return {
      game: g.name,
      fps: computed
    };
  });

  const width = container.clientWidth;
  const isSmallScreen = width < 420;
  const height = container.clientHeight || 220;
  const margin = { top: 10, right: 35, bottom: 25, left: isSmallScreen ? 100 : 140 };
  const innerW = width - margin.left - margin.right;
  const innerH = height - margin.top - margin.bottom;

  container.innerHTML = '';
  const svg = d3.select(container).append('svg')
    .attr('width', width)
    .attr('height', height);

  const g = svg.append('g').attr('transform', `translate(${margin.left},${margin.top})`);

  const maxVal = Math.max(160, d3.max(data, d => d.fps) * 1.15);
  const x = d3.scaleLinear().domain([0, maxVal]).range([0, innerW]);
  const y = d3.scaleBand().domain(data.map(d => d.game)).range([0, innerH]).padding(0.3);

  // X Axis Grid
  g.append('g')
    .attr('transform', `translate(0, ${innerH})`)
    .call(d3.axisBottom(x).ticks(5).tickSize(-innerH).tickFormat(d => `${d} fps`))
    .call(g => g.select('.domain').remove())
    .call(g => g.selectAll('.tick line').attr('stroke', 'rgba(255,255,255,0.05)'))
    .call(g => g.selectAll('.tick text').attr('fill', '#64748b').style('font-size', '10px'));

  // Y Axis
  g.append('g')
    .call(d3.axisLeft(y).tickSize(0))
    .call(g => g.select('.domain').remove())
    .call(g => g.selectAll('.tick text')
      .attr('fill', '#94a3b8')
      .style('font-family', 'var(--font-body)')
      .style('font-weight', '500')
      .style('font-size', '11px')
    );

  // 60 & 144 FPS Reference Lines
  [60, 144].forEach(val => {
    if (val < maxVal) {
      g.append('line')
        .attr('x1', x(val)).attr('x2', x(val))
        .attr('y1', 0).attr('y2', innerH)
        .attr('stroke', val === 60 ? 'rgba(245, 158, 11, 0.25)' : 'rgba(200, 150, 80, 0.25)')
        .attr('stroke-dasharray', '2 3');
      
      g.append('text')
        .attr('x', x(val) + 3)
        .attr('y', 9)
        .attr('fill', val === 60 ? '#f59e0b' : '#e5a968')
        .attr('font-size', '9px')
        .attr('font-weight', '600')
        .text(`${val} FPS`);
    }
  });

  const tooltip = d3.select('#d3-tooltip');

  // Clean solid bronze bars
  g.selectAll('.fps-bar')
    .data(data)
    .join('rect')
    .attr('class', 'fps-bar')
    .attr('y', d => y(d.game))
    .attr('height', y.bandwidth())
    .attr('rx', 3)
    .attr('fill', '#c89650')
    .attr('x', 0)
    .attr('width', 0)
    .on('mouseenter', (event, d) => {
      tooltip.style('opacity', 1)
        .html(`<strong>${d.game}</strong><br><span style="color:#e5a968;font-weight:600;">${d.fps} FPS</span> moyens (${resolution})`)
        .style('left', (event.pageX + 10) + 'px')
        .style('top', (event.pageY - 28) + 'px');
    })
    .on('mousemove', (event) => {
      tooltip.style('left', (event.pageX + 10) + 'px').style('top', (event.pageY - 28) + 'px');
    })
    .on('mouseleave', () => { tooltip.style('opacity', 0); })
    .transition()
    .duration(400)
    .ease(d3.easeCubicOut)
    .attr('width', d => x(d.fps));

  // FPS Value labels
  g.selectAll('.fps-label')
    .data(data)
    .join('text')
    .attr('class', 'fps-label')
    .attr('y', d => y(d.game) + y.bandwidth() / 2 + 4)
    .attr('x', d => x(d.fps) + 6)
    .attr('fill', '#e2e8f0')
    .style('font-family', 'var(--font-heading)')
    .style('font-weight', '600')
    .style('font-size', '11px')
    .text(d => `${d.fps}`);
}

function renderBudgetChart() {
  const container = document.getElementById('budget-chart-container');
  if (!container || container.clientWidth === 0) return;

  const data = Object.keys(currentSelection).map(cat => {
    const item = HARDWARE_DB[cat].find(i => i.id === currentSelection[cat]);
    const labels = {
      cpu: 'CPU', gpu: 'GPU', mobo: 'Carte Mère', ram: 'RAM',
      ssd: 'SSD', cooler: 'Cooling', psu: 'Alim', case: 'Boîtier',
      monitor: 'Écran', keyboard: 'Clavier', mouse: 'Souris',
      headset: 'Casque', os: 'OS / Service'
    };
    return {
      name: labels[cat] || cat,
      price: item ? getMerchantItemPrice(item.price, preferredMerchant, cat) : 0
    };
  }).filter(d => d.price > 0);

  const total = d3.sum(data, d => d.price);
  const width = container.clientWidth;
  const isSmallScreen = width < 420;
  const height = container.clientHeight || 190;
  const radius = isSmallScreen 
    ? Math.min(width * 0.52, height) / 2 - 6 
    : Math.min(width, height) / 2 - 10;

  const donutX = isSmallScreen ? width * 0.31 : width / 2.9;

  container.innerHTML = '';
  const svg = d3.select(container).append('svg')
    .attr('width', width)
    .attr('height', height)
    .append('g')
    .attr('transform', `translate(${donutX}, ${height / 2})`);

  // Professional balanced palette
  const color = d3.scaleOrdinal()
    .domain(data.map(d => d.name))
    .range(['#c89650', '#14b8a6', '#e5a968', '#10b981', '#f59e0b', '#a8a29e', '#64748b', '#475569', '#38bdf8', '#d97706', '#06b6d4', '#8b5cf6', '#ec4899']);

  const pie = d3.pie().value(d => d.price).sort(null);
  const arc = d3.arc().innerRadius(radius * 0.6).outerRadius(radius);
  const tooltip = d3.select('#d3-tooltip');

  const arcs = svg.selectAll('.arc')
    .data(pie(data))
    .join('g')
    .attr('class', 'arc');

  arcs.append('path')
    .attr('d', arc)
    .attr('fill', d => color(d.data.name))
    .attr('stroke', '#171b22')
    .attr('stroke-width', 2)
    .style('cursor', 'pointer')
    .on('mouseenter', (event, d) => {
      const pct = Math.round((d.data.price / total) * 100);
      tooltip.style('opacity', 1)
        .html(`<strong>${d.data.name}</strong><br>${d.data.price} € (${pct}% du total)`)
        .style('left', (event.pageX + 10) + 'px')
        .style('top', (event.pageY - 28) + 'px');
    })
    .on('mousemove', (event) => {
      tooltip.style('left', (event.pageX + 10) + 'px').style('top', (event.pageY - 28) + 'px');
    })
    .on('mouseleave', () => { tooltip.style('opacity', 0); });

  // Center Text in Donut
  svg.append('text')
    .attr('text-anchor', 'middle')
    .attr('dy', '-0.2em')
    .attr('fill', '#64748b')
    .style('font-size', '10px')
    .style('font-weight', '600')
    .text('TOTAL');

  svg.append('text')
    .attr('text-anchor', 'middle')
    .attr('dy', '1.05em')
    .attr('fill', '#ffffff')
    .style('font-family', 'var(--font-heading)')
    .style('font-weight', '700')
    .style('font-size', '14px')
    .text(`${total} €`);

  // Side Legend
  const legendX = isSmallScreen ? width * 0.60 : width / 1.55;
  const legendG = d3.select(container).select('svg')
    .append('g')
    .attr('transform', `translate(${legendX}, 14)`);

  data.forEach((d, i) => {
    const row = legendG.append('g').attr('transform', `translate(0, ${i * 19})`);
    row.append('rect')
      .attr('width', 9).attr('height', 9).attr('rx', 2)
      .attr('fill', color(d.name));
    row.append('text')
      .attr('x', 14).attr('y', 8)
      .attr('fill', '#94a3b8')
      .style('font-size', '10px')
      .style('font-family', 'var(--font-body)')
      .text(`${d.name} : ${Math.round((d.price / total) * 100)}%`);
  });
}

window.addEventListener('DOMContentLoaded', () => {
  initSelectOptions();

  // Restore affiliate configuration
  try {
    const storedAff = localStorage.getItem('dazl-affiliate-config');
    if (storedAff) {
      affiliateConfig = { ...affiliateConfig, ...JSON.parse(storedAff) };
    }
  } catch (e) {}

  if (window.smGetUrlParam) {
    const urlAmz = window.smGetUrlParam('amz_tag');
    const urlSub = window.smGetUrlParam('sub_id');
    if (urlAmz) affiliateConfig.amazonTag = urlAmz;
    if (urlSub) affiliateConfig.subId = urlSub;
  }
  updateAffiliateUIState();

  // Restore URL parameters if present
  if (window.smGetUrlParam) {
    const savedVendor = window.smGetUrlParam('vendor');
    if (savedVendor && VENDORS[savedVendor]) {
      preferredMerchant = savedVendor;
      const sel = document.getElementById('select-global-merchant');
      if (sel) sel.value = savedVendor;
    }
    Object.keys(currentSelection).forEach(cat => {
      const saved = window.smGetUrlParam(cat);
      if (saved && HARDWARE_DB[cat].some(i => i.id === saved)) {
        currentSelection[cat] = saved;
      }
    });
    const savedTab = window.smGetUrlParam('tab');
    if (savedTab) switchView(savedTab);
    const savedBudget = window.smGetUrlParam('target_budget');
    if (savedBudget) {
      const parsedB = parseInt(savedBudget, 10);
      if (!isNaN(parsedB) && parsedB > 0) {
        targetBudget = parsedB;
        const bInput = document.getElementById('input-target-budget');
        if (bInput) bInput.value = parsedB;
        highlightActiveBudgetPill(parsedB);
      }
    }
  }

  updateTargetBudgetBtnLabel(targetBudget);
  updateSelectElements();
  recalculateAll();

  // Debounced Responsive chart resize observer
  let roTimer = null;
  const ro = new ResizeObserver(() => {
    if (roTimer) clearTimeout(roTimer);
    roTimer = setTimeout(() => {
      window.requestAnimationFrame(() => {
        renderFpsChart();
        renderBudgetChart();
      });
    }, 100);
  });
  const chartFps = document.getElementById('fps-chart-container');
  const chartBudget = document.getElementById('budget-chart-container');
  if (chartFps) ro.observe(chartFps);
  if (chartBudget) ro.observe(chartBudget);
});
